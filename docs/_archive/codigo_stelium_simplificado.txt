<script>
// Executar imediatamente sem DOMContentLoaded
(function() {
    console.log('🌟 Stelium Theme: Iniciando...');

    // Função para aplicar tema
    function aplicarTema() {
        var loginCard = document.querySelector('.box-shadow.flex.min-w-\\[300px\\].flex-col.rounded-md');

        if (!loginCard) {
            console.log('⚠️ Card não encontrado ainda, tentando novamente...');
            return false;
        }

        console.log('✅ Card encontrado!');

        // 1. Modificar título "Sign In" para "Bem-vindo"
        var signInTitle = loginCard.querySelector('p.p-4.text-xl.font-bold');
        if (signInTitle && signInTitle.textContent.includes('Sign')) {
            signInTitle.textContent = 'Bem-vindo';
            console.log('✓ Título modificado');

            // 2. Adicionar subtítulo
            var subtitle = document.createElement('p');
            subtitle.style.cssText = 'text-align:center;font-size:0.875rem;color:rgba(255,255,255,0.75);margin:0 0 16px 0;padding:0 32px;position:relative;z-index:1;';
            subtitle.textContent = 'Acesse sua conta para continuar';
            signInTitle.parentNode.insertBefore(subtitle, signInTitle.nextSibling);
            console.log('✓ Subtítulo adicionado');
        }

        // 3. Esconder "Powered by"
        var poweredBy = document.querySelector('.text-sm.font-normal');
        if (poweredBy && poweredBy.textContent.includes('Powered')) {
            poweredBy.style.display = 'none';
            console.log('✓ Powered by escondido');
        }

        console.log('✅ Stelium Theme aplicado com sucesso!');
        return true;
    }

    // Tentar aplicar imediatamente
    if (!aplicarTema()) {
        // Se falhar, tentar depois de 100ms
        setTimeout(aplicarTema, 100);
    }
})();
</script>
